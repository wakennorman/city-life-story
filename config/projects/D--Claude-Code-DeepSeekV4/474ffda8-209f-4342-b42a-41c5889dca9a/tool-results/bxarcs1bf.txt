  1053	/** 街头每日事件判定 */
  1054	function rollStreetEvent(state) {
  1055	  // 基础 18% 触发率，已存在待弹事件时不重复触发
  1056	  if (state._pendingEvent) return;
  1057	  const baseChance = 0.18;
  1058	  // 健康差或债务高时提高触发率
  1059	  let mod = 0;
  1060	  if (state.status.health < 50) mod += 0.1;
  1061	  if (state.resources.debt > 3000) mod += 0.05;
  1062	  if (state.needs.happiness < 30) mod += 0.05;
  1063	  if (Math.random() < baseChance + mod) {
  1064	    queueRandomEvent(state, "street");
  1065	  }
  1066	}
  1067	
  1068	/** 职场每日事件判定 */
  1069	function rollCorporateEvent(state) {
  1070	  if (state._pendingEvent) return;
  1071	  const baseChance = 0.22;
  1072	  let mod = 0;
  1073	  if (state.player.corporate.risk > 50) mod += 0.1;
  1074	  if (state.player.corporate.popularity < 30) mod += 0.05;
  1075	  if (state.player.corporate.upwardMgmt < 20) mod += 0.05;
  1076	  if (Math.random() < baseChance + mod) {
  1077	    queueRandomEvent(state, "corporate");
  1078	  }
  1079	}

  1081	/** 把一个随机事件塞进待弹队列 */
  1082	function queueRandomEvent(state, phase) {
  1083	  const pool = RANDOM_EVENTS.filter((e) => e.phase === phase);
  1084	  if (pool.length === 0) return;
  1085	
  1086	  // 过滤掉不满足条件的事件
  1087	  const eligible = pool.filter((e) => !e.conditions || e.conditions(state));
  1088	  if (eligible.length === 0) return;
  1089	
  1090	  const evt = eligible[Math.floor(Math.random() * eligible.length)];
  1091	  // 使用唯一事件ID替代引用比较，避免引用失效导致事件卡住
  1092	  state._pendingEvent = evt;
  1093	  state._pendingEventId = evt.id;
  1094	  // 触发延迟到 render 阶段弹（避免在 tick 内部阻塞）
  1095	  setTimeout(() => {
  1096	    const s = StateManager.getState();
  1097	    if (s._pendingEvent && s._pendingEventId === evt.id) {
  1098	      showEventModal(evt);
  1099	      // 超时兜底：10秒后若事件仍未清理，强制清除（玩家可能忽略了弹窗）
  1100	      s._pendingEventTimeout = setTimeout(() => {
  1101	        if (s._pendingEventId === evt.id) {
  1102	          s._pendingEvent = null;
  1103	          s._pendingEventId = null;
  1104	          document.querySelector(".modal-overlay.event-modal")?.remove();
  1105	          console.warn("Event timed out, auto-cleaned:", evt.id);
  1106	        }
  1107	      }, 10000);
  1108	    }
  1109	  }, 50);
  1110	}

  1116	/**
  1117	 * 渲染并展示一个事件模态框
  1118	 * @param {Object} evt - 事件对象
  1119	 */
  1120	function showEventModal(evt) {
  1121	  // 先卸掉任何旧弹窗
  1122	  document.querySelector(".modal-overlay")?.remove();
  1123	
  1124	  // 构建选项HTML
  1125	  const choicesHtml = evt.choices
  1126	    .map((ch, i) => {
  1127	      const hintStr = ch.hint
  1128	        ? `<div class="choice-hint">${ch.hint}</div>`
  1129	        : "";
  1130	      // 检查现金是否够
  1131	      let disabled = false;
  1132	      if (ch.cost && StateManager.getState().resources.cash < ch.cost) {
  1133	        disabled = true;
  1134	      }
  1135	      const costTag = ch.cost
  1136	        ? ` <span style="color:var(--warning);font-size:11px;">需 ¥${ch.cost}</span>`
  1137	        : "";
  1138	      return `
  1139	        <button class="event-choice ${disabled ? "disabled" : ""}" data-idx="${i}" ${disabled ? "disabled" : ""}>
  1140	          <div class="choice-main">${ch.text}${costTag}</div>
  1141	          ${hintStr}
  1142	        </button>
  1143	      `;
  1144	    })
  1145	    .join("");
  1146	
  1147	  const overlay = document.createElement("div");
  1148	  overlay.className = "modal-overlay event-modal";
  1149	  overlay.innerHTML = `
  1150	    <div class="modal-box event-box">
  1151	      <div class="event-header">
  1152	        <div class="event-icon">${evt.icon}</div>
  1153	        <h2 class="event-title">${evt.title}</h2>
  1154	      </div>
  1155	      <p class="event-story">${evt.story}</p>
  1156	      <div class="event-choices">${choicesHtml}</div>
  1157	      <div style="text-align:center;margin-top:8px;font-size:10px;color:var(--text-muted);">
  1158	        ⏰ 这是一个会影响数值的选择，请谨慎决定
  1159	      </div>
  1160	    </div>
  1161	  `;
  1162	  document.body.appendChild(overlay);
  1163	
  1164	  // 绑定选项点击
  1165	  overlay.querySelectorAll(".event-choice").forEach((btn) => {
  1166	    btn.addEventListener("click", () => {
  1167	      const idx = parseInt(btn.dataset.idx);
  1168	      const choice = evt.choices[idx];
  1169	      if (!choice) return;
  1170	      const state = StateManager.getState();
  1171	      try {
  1172	        if (typeof choice.apply === "function") {
  1173	          choice.apply(state);
  1174	        }
  1175	      } catch (e) {
  1176	        console.error("Event choice apply error:", e);
  1177	      }
  1178	      // 清掉待弹事件（三字段全部清理）
  1179	      state._pendingEvent = null;
  1180	      state._pendingEventId = null;
  1181	      if (state._pendingEventTimeout) {
  1182	        clearTimeout(state._pendingEventTimeout);
  1183	        state._pendingEventTimeout = null;
  1184	      }
  1185	      // 关闭弹窗 + 重新渲染
  1186	      document.body.removeChild(overlay);
  1187	      renderAll();
  1188	    });
  1189	  });
  1190	
  1191	  // 点击遮罩关闭也清理事件悬挂
  1192	  overlay.addEventListener("click", (e) => {
  1193	    if (e.target === overlay) {
  1194	      const s = StateManager.getState();
  1195	      s._pendingEvent = null;
  1196	      s._pendingEventId = null;
  1197	      if (s._pendingEventTimeout) {
  1198	        clearTimeout(s._pendingEventTimeout);
  1199	        s._pendingEventTimeout = null;
  1200	      }
  1201	      document.body.removeChild(overlay);
  1202	      renderAll();
  1203	    }
  1204	  });
  1205	}
