    12	/** 设置属性条 */
    13	function setStatBar(id, value, cssClass) {
    14	  const wrap = document.getElementById(id);
    15	  if (!wrap) return;
    16	  const bar = wrap.querySelector(".stat-bar");
    17	  const valEl = wrap.querySelector(".stat-value");
    18	  if (bar) {
    19	    bar.style.width = Math.max(0, Math.min(100, value)) + "%";
    20	    bar.className = "stat-bar " + cssClass;
    21	  }
    22	  if (valEl) {
    23	    valEl.textContent = Math.round(value);
    24	  }
    25	}

    27	/** 描述装备效果 */
    28	function describeItemEffects(itemDef) {
    29	  if (!itemDef || !itemDef.effects) return "";
    30	  const parts = [];
    31	  for (const [key, val] of Object.entries(itemDef.effects)) {
    32	    if (key === "capacity") parts.push(`容量+${val}`);
    33	    else if (key === "injury") parts.push(`受伤率${val < 0 ? val : "+" + val}`);
    34	    else if (key === "illness")
    35	      parts.push(`生病率${val < 0 ? val : "+" + val}`);
    36	    else if (key === "fatigue") parts.push(`疲劳${val < 0 ? val : "+" + val}`);
    37	    else if (typeof val === "number")
    38	      parts.push(`${key}${val > 0 ? "+" + val : val}`);
    39	  }
    40	  return parts.join(", ");
    41	}

    43	/** 检查工作需求是否满足 */
    44	function checkJobRequirements(job, state) {
    45	  const reqs = job.requirements || {};
    46	  const p = state.player;
    47	  const s = state.skills;
    48	
    49	  if (reqs.minAge && p.age < reqs.minAge) return `需要年满${reqs.minAge}岁`;
    50	  if (reqs.maxAge && p.age > reqs.maxAge) return `年龄超过${reqs.maxAge}岁限制`;
    51	  if (reqs.physique && p.physique < reqs.physique)
    52	    return `体质不足 (需要${reqs.physique})`;
    53	  if (reqs.intelligence && p.intelligence < reqs.intelligence)
    54	    return `智力不足 (需要${reqs.intelligence})`;
    55	  if (reqs.agility && p.agility < reqs.agility)
    56	    return `敏捷不足 (需要${reqs.agility})`;
    57	  if (reqs.mental && p.mental < reqs.mental)
    58	    return `心智不足 (需要${reqs.mental})`;
    59	  if (reqs.cooking && s.cooking.level < reqs.cooking)
    60	    return `烹饪技能不足 (需要${reqs.cooking})`;
    61	  if (reqs.repair && s.repair.level < reqs.repair)
    62	    return `维修技能不足 (需要${reqs.repair})`;
    63	  if (reqs.sales && s.sales.level < reqs.sales)
    64	    return `销售技能不足 (需要${reqs.sales})`;
    65	  if (reqs.english && s.english.level < reqs.english)
    66	    return `英语技能不足 (需要${reqs.english})`;
    67	
    68	  return null; // 通过
    69	}

    71	/** 估算工作收入 */
    72	function estimateJobPay(job, state) {
    73	  // 模拟3次取平均
    74	  let total = 0;
    75	  for (let i = 0; i < 10; i++) {
    76	    total += job.payCalc(state);
    77	  }
    78	  return Math.floor(total / 10);
    79	}

   189	/** 初始化各地商品价格 */
   190	function initializePrices() {
   191	  const state = StateManager.getState();
   192	  for (const locKey of Object.keys(LOCATIONS)) {
   193	    state.trade.goodsPrices[locKey] = {};
   194	    const loc = LOCATIONS[locKey];
   195	    for (const good of GOODS) {
   196	      let price = good.basePrice;
   197	      if (loc.priceMod && loc.priceMod[good.id]) {
   198	        price *= loc.priceMod[good.id];
   199	      }
   200	      // 随机波动 ±10%
   201	      price *= 0.9 + Math.random() * 0.2;
   202	      price = Math.round(price * 100) / 100;
   203	      state.trade.goodsPrices[locKey][good.id] = price;
   204	    }
   205	  }
   206	  state.trade.lastPriceUpdate = state.player.day;
   207	}

   209	// ====== 可用行动列表 ======
   210	function getAvailableActions(state) {

   212	
   213	  // 提升为函数级变量（用 let，块内可重新指向用 const）
   214	  const locKey = state.trade.currentLocation;

   220	    const jobIds = getJobsAtLocation(locKey);
   221	    for (const jobId of jobIds) {
   222	      const job = getJobById(jobId);
   223	      if (!job) continue;
   224	      const reqFail = checkJobRequirements(job, state);
   225	      const payEstimate = reqFail ? null : estimateJobPay(job, state);
   226	
   227	      actions.push({
   228	        id: "job_" + jobId,
   229	        name: job.name,
   230	        desc: job.desc,
   231	        icon: job.icon,
   232	        payEstimate: payEstimate
   233	          ? `${payEstimate - (job.startupCost || 0)}~${payEstimate + Math.floor(payEstimate * 0.3)}`
   234	          : null,
   235	        costEstimate: job.startupCost || null,
   236	        reqFail,
   237	        disabled:
   238	          !!reqFail ||
   239	          (job.startupCost && state.resources.cash < job.startupCost
   240	            ? `启动资金不足(需¥${job.startupCost})`
   241	            : null) ||
   242	          null,
   243	        handler: () => {
   244	          const finalReq = checkJobRequirements(job, state);
   245	          if (finalReq) {
   246	            StateManager.addMessage(`⚠️ ${finalReq}`, "warning");
   247	            return;
   248	          }

   250	            StateManager.addMessage(
   251	              `⚠️ 启动资金不足，需要 ¥${job.startupCost}`,
   252	              "warning",
   253	            );
   254	            return;
   255	          }
   256	          doStreetJob(job);
   257	        },
   258	      });
   259	    }
   260	
   261	    // 如果没有可用工作
   262	    if (jobIds.length === 0) {
   263	      actions.push({
   264	        id: "no_jobs",
   265	        name: "当前地点没有工作机会",

   285	    // 零售买卖 — 在任何有价格数据的地点都可以买卖

   287	      state.trade.goodsPrices[locKey] &&
   288	      Object.keys(state.trade.goodsPrices[locKey]).length > 0
   289	    ) {
   290	      actions.push({
   291	        id: "trade_header",
   292	        name: "🛒 买卖商品（转到交易Tab）",
   293	        desc: "查看当前市场价格，低买高卖赚取差价。",

  1108	/**
  1109	 * 技能加权收入倍率
  1110	 * 根据工作类型，玩家相关技能等级会提升收入。
  1111	 * 每10级技能提升约5%收入，上限+50%（Lv.100）。
  1112	 * 减少纯RNG依赖，奖励技能投资。
  1113	 */
  1114	function getSkillPayBonus(jobId, state) {
  1115	  const s = state.skills;
  1116	  const skillMap = {
  1117	    // 维修类 -> repair 技能
  1118	    repair_service: { key: "repair", weight: 1.0 },
  1119	    skilled_labor_construction: { key: "repair", weight: 0.6 },
  1120	    // 摆摊/销售类 -> sales 技能
  1121	    street_vending_food: { key: "sales", weight: 0.8 },
  1122	    street_vending_goods: { key: "sales", weight: 1.0 },
  1123	    food_stall: { key: "cooking", weight: 0.9 },
  1124	    // 配送类 -> agility
  1125	    delivery_rider: { key: "agility", weight: 0.7 },
  1126	    package_delivery: { key: "agility", weight: 0.6 },
  1127	    // 仓库/体力类 -> physique
  1128	    warehouse_worker: { key: "physique", weight: 0.5 },
  1129	    manual_labor_construction: { key: "physique", weight: 0.6 },
  1130	    factory_work_assembly: { key: "physique", weight: 0.5 },
  1131	    // 保安 -> physique
  1132	    security_guard: { key: "physique", weight: 0.3 },
  1133	    // 理发 -> repair(手艺)
  1134	    barber: { key: "repair", weight: 0.5 },
  1135	    // 清洁 -> physique
  1136	    cleaning_service: { key: "physique", weight: 0.3 },
  1137	    // 家教 -> intelligence
  1138	    tutoring: { key: "intelligence", weight: 0.8 },
  1139	    // 表演 -> sales(吸引打赏)
  1140	    street_performer: { key: "sales", weight: 0.5 },
  1141	  };
  1142	
  1143	  const mapping = skillMap[jobId];
  1144	  if (!mapping) return 1.0;
  1145	
  1146	  // 兼容属性键和技能键
  1147	  let effectiveLevel = 0;
  1148	  const attrKeys = ["physique", "intelligence", "agility", "mental"];
  1149	  if (attrKeys.includes(mapping.key)) {
  1150	    effectiveLevel = state.player[mapping.key] || 0;
  1151	  } else if (s[mapping.key]) {
  1152	    effectiveLevel = s[mapping.key].level || 0;
  1153	  }
  1154	
  1155	  // 每10级 +5%收入，权重调节
  1156	  const bonus = (effectiveLevel / 10) * 0.05 * mapping.weight;
  1157	  return 1.0 + Math.max(0, Math.min(0.5, bonus));
  1158	}

  1162	/** 消耗行动力，推进时间。每次行动调用此函数替代旧的 advanceTimeSlot() */
  1163	function consumeAP(cost) {
  1164	  const state = StateManager.getState();
  1165	  state.player.actionPoints = Math.max(0, state.player.actionPoints - cost);
  1166	
  1167	  // 更新显示用的时段
  1168	  const pct = state.player.actionPoints / state.player.maxActionPoints;
  1169	  if (pct > 0.66) state.player.timeSlot = "morning";
  1170	  else if (pct > 0.33) state.player.timeSlot = "afternoon";
  1171	  else state.player.timeSlot = "evening";
  1172	
  1173	  // Per-action 随机事件判定 (6%基础概率)
  1174	  if (!state._pendingEvent) {
  1175	    const phase = state.player.phase === "corporate" ? "corporate" : "street";
  1176	    if (Math.random() < 0.06) {
  1177	      if (typeof queueRandomEvent === "function")
  1178	        queueRandomEvent(state, phase);
  1179	    }
  1180	  }
  1181	
  1182	  // 行动力耗尽 → 结束一天
  1183	  if (state.player.actionPoints <= 0) {
  1184	    endDay();
  1185	  }
  1186	}

  1193	function endDay() {
  1194	  const state = StateManager.getState();
  1195	
  1196	  state.player.day++;
  1197	  // 重置行动力
  1198	  state.player.actionPoints = state.player.maxActionPoints;
  1199	  state.player.timeSlot = "morning";
  1200	
  1201	  // 使用需求系统模块
  1202	  if (typeof applyNeedsDecay === "function") applyNeedsDecay(state);

  1204	    state.needs.hunger = Math.max(0, state.needs.hunger - 18);

  1207	  // 睡眠恢复（基础25 + 住所加成）

  1210	    0,

  1217	  state.needs.happiness = Math.max(
  1218	    0,
  1219	    Math.min(100, state.needs.happiness - 3 + (house.happinessBonus || 0)),

  1221	
  1222	  // 扣房租
  1223	  if (house.rent > 0) {
